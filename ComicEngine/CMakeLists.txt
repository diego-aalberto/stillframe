cmake_minimum_required(VERSION 3.20)
project(ComicEngine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(comic_core src/core.cpp src/demo.cpp src/anime_demo.cpp)
target_include_directories(comic_core PUBLIC src)
add_executable(ComicTests tests/core_tests.cpp)
target_link_libraries(ComicTests PRIVATE comic_core)
enable_testing()
add_test(NAME core COMMAND ComicTests)
if(WIN32)
  foreach(app IN ITEMS ComicPlayer ComicStudio)
    add_executable(${app} WIN32 src/app.cpp src/platform.cpp src/render.cpp)
    target_compile_definitions(${app} PRIVATE UNICODE _UNICODE)
    target_link_libraries(${app} PRIVATE comic_core opengl32 glu32 gdi32 comdlg32 comctl32 ole32 windowscodecs uuid shell32)
    if(MINGW)
      target_link_options(${app} PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
  endforeach()
  target_compile_definitions(ComicStudio PRIVATE COMIC_EDITOR)
  add_executable(StoryGame WIN32 src/story_game.cpp src/story_art.cpp src/platform.cpp src/render.cpp)
  target_compile_definitions(StoryGame PRIVATE UNICODE _UNICODE)
  target_link_libraries(StoryGame PRIVATE comic_core opengl32 glu32 gdi32 comdlg32 ole32 windowscodecs uuid shell32 winhttp gdiplus winmm)
  if(MINGW)
    target_link_options(StoryGame PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()
