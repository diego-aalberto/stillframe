cmake_minimum_required(VERSION 3.20)
project(Stillframe VERSION 2.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /wd4100)
else()
  add_compile_options(-Wall -Wextra -Wno-unused-parameter -Wno-maybe-uninitialized -Wno-dangling-reference)
endif()

find_package(Threads REQUIRED)

file(GLOB SF_CORE_SOURCES CONFIGURE_DEPENDS core/*.cpp)
add_library(sf_core STATIC ${SF_CORE_SOURCES})
target_include_directories(sf_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(sf_core PUBLIC Threads::Threads)
target_compile_definitions(sf_core PRIVATE SF_ASSETS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
if(WIN32)
  target_link_libraries(sf_core PUBLIC ws2_32)
endif()

file(GLOB SF_RENDER_SOURCES CONFIGURE_DEPENDS render/*.cpp)
add_library(sf_render STATIC ${SF_RENDER_SOURCES})
target_link_libraries(sf_render PUBLIC sf_core)

file(GLOB SF_TEST_SOURCES CONFIGURE_DEPENDS tests/*.cpp)
add_executable(sf_tests ${SF_TEST_SOURCES})
target_link_libraries(sf_tests PRIVATE sf_core sf_render)
target_compile_definitions(sf_tests PRIVATE SF_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")

enable_testing()
add_test(NAME sf_tests COMMAND sf_tests WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})

# The command-line program. Windowed play needs SDL3 (+ SDL3_mixer for sound); everything else works without it.
file(GLOB SF_APP_SOURCES CONFIGURE_DEPENDS app/*.cpp)
list(FILTER SF_APP_SOURCES EXCLUDE REGEX ".*/sdl_.*\\.cpp$")
add_executable(stillframe ${SF_APP_SOURCES})
target_link_libraries(stillframe PRIVATE sf_render)
find_package(SDL3 CONFIG QUIET)
if(SDL3_FOUND AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/app/sdl_app.cpp)
  target_sources(stillframe PRIVATE app/sdl_app.cpp)
  target_compile_definitions(stillframe PRIVATE SF_WITH_SDL=1)
  target_link_libraries(stillframe PRIVATE SDL3::SDL3)
  find_package(SDL3_mixer CONFIG QUIET)
  if(SDL3_mixer_FOUND)
    target_compile_definitions(stillframe PRIVATE SF_WITH_MIXER=1)
    target_link_libraries(stillframe PRIVATE SDL3_mixer::SDL3_mixer)
  endif()
  message(STATUS "Stillframe: window support enabled (SDL3 found)")
else()
  message(STATUS "Stillframe: SDL3 not found - building without a window (CLI tools only)")
endif()
